# 旧:

# 渲染 placeholders (跟"反讽" 联合, 跟"诚实修正" 联合)
KALLAX_ROOT_CALC="$(cd "${ONRAMP_DIR}/../.." && pwd)"

# 跑 scan 拿数据 (跟"反讽" 联合, 不靠 LLM)
SCAN_DATA=$(bash "${ONRAMP_DIR}/lib/scan.sh" "${PROJECT_PATH}" 2>/dev/null || echo "{}")
loc=$(echo "${SCAN_DATA}" | jq -r '.loc // "?"' 2>/dev/null || echo "?")
files=$(echo "${SCAN_DATA}" | jq -r '.files // "?"' 2>/dev/null || echo "?")
modules=$(echo "${SCAN_DATA}" | jq -r '.modules // "?"' 2>/dev/null || echo "?")
has_claude_md=$(echo "${SCAN_DATA}" | jq -r '.has_claude_md // false' 2>/dev/null || echo "false")
has_readme=$(echo "${SCAN_DATA}" | jq -r '.has_readme // false' 2>/dev/null || echo "false")
language_mix=$(echo "${SCAN_DATA}" | jq -r '.language_mix // "unknown"' 2>/dev/null || echo "unknown")
git_log_days=$(echo "${SCAN_DATA}" | jq -r '.git_log_days // 0' 2>/dev/null || echo "0")

# 渲染 expert output (跟"反讽" 联合, 跟"诚实修正" 联合)
expert_count=$(echo "${CHOICE_JSON}" | jq -r '.experts | length')
experts_list=$(echo "${CHOICE_JSON}" | jq -r '.experts | join(", ")')

# 读 skill 文档内容作为 expert output (跟 B4 联合)
expert_outputs=""
i=0
for expert in $(echo "${CHOICE_JSON}" | jq -r '.experts[]'); do
  i=$((i+1))
  skill_path="/Users/chenchen/.claude/skills/kallax/default/${expert}.md"
  if [[ ! -f "${skill_path}" ]]; then
    skill_path="/Users/chenchen/.claude/skills/kallax/extended/${expert}.md"
  fi
  if [[ -f "${skill_path}" ]]; then
    skill_name=$(grep -m1 "^name:" "${skill_path}" 2>/dev/null | sed 's/^name:[[:space:]]*//' || echo "${expert}")
    skill_desc=$(grep -m1 "^description:" "${skill_path}" 2>/dev/null | sed 's/^description:[[:space:]]*//' || echo "")
    expert_outputs="${expert_outputs}
### ${i}. ${skill_name}
**角色**: ${expert}
**Skill 路径**: ${skill_path}
**描述**: ${skill_desc}

"
  fi
done

# sed replace placeholders (跟"反讽" 联合, 跟"诚实修正" 联合)
sed -i '' "s|{{project}}|${project_safe}|g" "${tmp_file}"
sed -i '' "s|{{date}}|${today}|g" "${tmp_file}"

sed -i '' "s|{{files}}|${files}|g" "${tmp_file}"
sed -i '' "s|{{modules}}|${modules}|g" "${tmp_file}"
sed -i '' "s|{{has_claude_md}}|${has_claude_md}|g" "${tmp_file}"
sed -i '' "s|{{has_readme}}|${has_readme}|g" "${tmp_file}"
sed -i '' "s|{{language_mix}}|${language_mix}|g" "${tmp_file}"
sed -i '' "s|{{git_log_days}}|${git_log_days}|g" "${tmp_file}"
sed -i '' "s|{{expert_count}}|${expert_count}|g" "${tmp_file}"
sed -i '' "s|{{experts_list}}|${experts_list}|g" "${tmp_file}"
sed -i '' "s|{{expert_output}}|见下方各专家分析|g" "${tmp_file}"
sed -i '' "s|{{expert_outputs}}|${expert_outputs}|g" "${tmp_file}"
